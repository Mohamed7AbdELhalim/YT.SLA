================================================================================
                         1. نظرة عامة على السكربت
================================================================================

هذا السكربت يقوم بـ:
─────────────────────
• تنظيف روابط يوتيوب من معاملات التتبع (?si=, &feature=, إلخ)
• تحويل روابط youtu.be إلى youtube.com
• تصنيف الروابط تلقائياً (وسم الموقع + وسم المسار + المجلد)
• استخراج التاريخ من الرابط وحساب رقم الأسبوع
• جلب عناوين الفيديوهات وقوائم التشغيل وصفحات الويب
• قراءة/كتابة ملفات JSON و TXT
• تحميل الفيديوهات باستخدام yt-dlp

الميزات الخاصة:
────────────────
• دعم المواقع غير يوتيوب (arxiv, medium, github, إلخ)
• جلب معلومات قوائم التشغيل (عدد الفيديوهات، المشاهدات)
• التحميل المتوازي (عدة فيديوهات في نفس الوقت)
• سجل تفصيلي لكل العمليات

def log(message: str, level: str = "INFO", log_file: Optional[Path] = None) -> None:
    """
    طباعة رسالة على الشاشة وحفظها في ملف السجل
    """
    # ─────────────────────────────────────────────────────────────────────────
    # المعاملات:
    #   message: النص المراد طباعته (مثل "تم التحميل بنجاح")
    #   level: مستوى الرسالة:
    #          - "INFO": معلومات عادية
    #          - "WARNING": تحذير
    #          - "ERROR": خطأ
    #          - "SUCCESS": نجاح
    #   log_file: مسار ملف السجل (اختياري)
    #
    # الخطوات:
    #   1. الحصول على التاريخ والوقت الحالي
    #   2. تحديد رمز المستوى (إيموجي)
    #   3. تنسيق السطر
    #   4. طباعة على الشاشة
    #   5. حفظ في ملف السجل (إذا تم تحديده)
    
    now = dt.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    # ↑ الحصول على الوقت الحالي بصيغة: 2024-01-15 14:30:45
    
    level_icons = {
        "INFO": "ℹ️",
        "WARNING": "⚠️",
        "ERROR": "❌",
        "SUCCESS": "✅",
    }
    icon = level_icons.get(level, "•")
    # ↑ اختيار الإيموجي المناسب للمستوى
    
    line = f"[{now}] {icon} [{level}] {message}"
    # ↑ تنسيق السطر: [2024-01-15 14:30:45] ✅ [SUCCESS] تم التحميل بنجاح
    
    print(line)
    # ↑ طباعة على الشاشة
    
    if log_file is not None:
        log_file.parent.mkdir(parents=True, exist_ok=True)
        # ↑ إنشاء المجلدات إذا لم تكن موجودة
        
        with log_file.open("a", encoding="utf-8") as f:
            f.write(line + "\n")
        # ↑ فتح الملف في وضع الإضافة (append) وكتابة السطر

def clean_url(url: str) -> str:
    """
    تنظيف الرابط من معاملات التتبع مع الحفاظ على المعاملات المهمة
    """
    # ─────────────────────────────────────────────────────────────────────────
    # مثال:
    #   الإدخال:  "https://youtube.com/watch?v=abc&si=123&feature=share"
    #   الإخراج:  "https://youtube.com/watch?v=abc"
    
    try:
        url = url.strip()
        # ↑ إزالة المسافات من البداية والنهاية
        
        if not url:
            return ''
        # ↑ إذا كان فارغاً، نرجع نصاً فارغاً
        
        parsed = urlparse(url)
        # ↑ تقسيم الرابط إلى أجزائه:
        #   parsed.scheme = "https"
        #   parsed.hostname = "youtube.com"
        #   parsed.path = "/watch"
        #   parsed.query = "v=abc&si=123&feature=share"
        
        hostname = parsed.hostname.lower() if parsed.hostname else ''
        # ↑ الحصول على اسم الموقع بأحرف صغيرة
        
        # ═══════════════════════════════════════════════════════════════════
        # تحويل youtu.be إلى youtube.com
        # ═══════════════════════════════════════════════════════════════════
        if hostname in ('youtu.be', 'www.youtu.be'):
            video_id = parsed.path.lstrip('/')
            # ↑ استخراج معرّف الفيديو من المسار
            #   youtu.be/abc123 → abc123
            
            if video_id:
                new_query = {'v': video_id}
                # ↑ إنشاء معاملات جديدة مع معرّف الفيديو
                
                old_params = parse_qs(parsed.query)
                for param in ['t', 'list', 'index']:
                    if param in old_params:
                        new_query[param] = old_params[param][0]
                # ↑ نقل المعاملات المهمة (وقت البداية، القائمة، الترتيب)
                
                return f"https://youtube.com/watch?{urlencode(new_query)}"
                # ↑ إرجاع الرابط الجديد بصيغة youtube.com
        
        # ═══════════════════════════════════════════════════════════════════
        # تصفية المعاملات
        # ═══════════════════════════════════════════════════════════════════
        query_params = parse_qs(parsed.query)
        # ↑ تحليل معاملات الاستعلام
        #   {'v': ['abc'], 'si': ['123'], 'feature': ['share']}
        
        filtered_params = {}
        
        for param, values in query_params.items():
            if param.lower() in TRACKING_PARAMS:
                continue
            # ↑ تخطي معاملات التتبع (si, feature, fbclid, إلخ)
            
            if 'youtube.com' in hostname or 'youtu.be' in hostname:
                if param.lower() in IMPORTANT_PARAMS:
                    filtered_params[param] = values[0]
            # ↑ لروابط يوتيوب: الحفاظ على المعاملات المهمة فقط
            
            else:
                filtered_params[param] = values[0]
            # ↑ للمواقع الأخرى: الحفاظ على كل المعاملات غير التتبعية
        
        # ═══════════════════════════════════════════════════════════════════
        # إعادة بناء الرابط
        # ═══════════════════════════════════════════════════════════════════
        new_query = urlencode(filtered_params) if filtered_params else ''
        
        cleaned = urlunparse((
            parsed.scheme,      # https
            parsed.netloc,      # youtube.com
            parsed.path.rstrip('/'),  # /watch (بدون / في النهاية)
            parsed.params,      # (فارغ عادة)
            new_query,          # v=abc
            ''                  # (بدون fragment)
        ))
        
        return cleaned
        
    except Exception as e:
        log(f"خطأ في تنظيف الرابط: {url} - {e}", "WARNING")
        return url.strip()

def classify_url_type(url: str) -> str:
    """
    تحديد نوع الرابط (فيديو، شورت، قائمة تشغيل، قناة، صفحة ويب)
    """
    # ─────────────────────────────────────────────────────────────────────────
    # أمثلة:
    #   youtube.com/shorts/abc → 'short'
    #   youtube.com/watch?v=abc → 'video'
    #   youtube.com/playlist?list=abc → 'playlist'
    #   youtube.com/@channel → 'channel'
    #   arxiv.org/abs/123 → 'webpage'
    
    url_lower = url.lower()
    
    try:
        parsed = urlparse(url)
        hostname = (parsed.hostname or '').lower().replace('www.', '')
        
        # تصنيف روابط يوتيوب
        if 'youtube.com' in hostname or 'youtu.be' in hostname:
            if '/shorts/' in url_lower:
                return 'short'
            if 'playlist?' in url_lower or 'list=' in url_lower:
                return 'playlist'
            if '/@' in url or '/channel/' in url_lower or '/c/' in url_lower:
                return 'channel'
            if 'watch?v=' in url_lower or 'youtu.be/' in url_lower:
                return 'video'
            return 'video'
        
        # المواقع الأخرى
        return 'webpage'
        
    except Exception:
        return 'other'


def get_site_tag(hostname: str) -> str:
    """
    استخراج وسم الموقع من اسم الدومين
    """
    # ─────────────────────────────────────────────────────────────────────────
    # أمثلة:
    #   www.youtube.com → '#youtube'
    #   arxiv.org → '#arxiv'
    #   medium.com → '#medium'
    
    clean_host = hostname.lower().replace('www.', '')
    
    if 'youtube.com' in clean_host or 'youtu.be' in clean_host:
        return '#youtube'
    
    base = clean_host.split('.')[0]
    # ↑ استخراج الجزء الأول من الدومين
    #   arxiv.org → arxiv
    
    return f'#{base}' if base else '#site'

def detect_date_from_url(url: str) -> Optional[dt.date]:
    """
    محاولة استخراج التاريخ من الرابط
    """
    # ─────────────────────────────────────────────────────────────────────────
    # يبحث في المسار والمعاملات عن أنماط تاريخ مثل:
    #   - 2024-01-15
    #   - 15/01/2024
    #   - 20240115
    #
    # إذا وجد تاريخ صالح، يرجعه ككائن date
    # إذا لم يجد، يرجع None
    
    try:
        parsed = urlparse(url)
        search_text = f"{parsed.path} {parsed.query}"
        # ↑ البحث في المسار والمعاملات معاً
        
        for pattern, order in DATE_PATTERNS:
            match = re.search(pattern, search_text)
            if match:
                values = {}
                for i, key in enumerate(order):
                    values[key] = int(match.group(i + 1))
                # ↑ استخراج القيم من التطابق
                
                year = values.get('year', 0)
                month = values.get('month', 0)
                day = values.get('day', 0)
                
                if 1 <= month <= 12 and 1 <= day <= 31 and 1900 <= year <= 2100:
                    try:
                        return dt.date(year, month, day)
                    except ValueError:
                        continue
                # ↑ التحقق من صحة القيم وإرجاع التاريخ
        
        return None
        
    except Exception:
        return None


def get_iso_week(date: dt.date) -> int:
    """
    حساب رقم الأسبوع حسب معيار ISO 8601
    """
    # ─────────────────────────────────────────────────────────────────────────
    # معيار ISO 8601:
    #   - الأسبوع يبدأ يوم الإثنين
    #   - الأسبوع الأول هو الذي يحتوي على 4 يناير
    #   - الأسابيع مرقمة من 1 إلى 52/53
    #
    # مثال:
    #   15 يناير 2024 → الأسبوع 3
    
    return date.isocalendar()[1]
    # ↑ isocalendar() يرجع (السنة، رقم الأسبوع، رقم اليوم)


def format_week_label(date: dt.date) -> str:
    """
    تنسيق تسمية الأسبوع
    """
    # ─────────────────────────────────────────────────────────────────────────
    # مثال:
    #   15 يناير 2024 → "الأسبوع 3 - 2024"
    
    week = get_iso_week(date)
    return f"الأسبوع {week} - {date.year}"

def fetch_youtube_video_title(url: str) -> Optional[str]:
    """
    جلب عنوان فيديو يوتيوب باستخدام oEmbed API
    """
    # ─────────────────────────────────────────────────────────────────────────
    # oEmbed API:
    #   - API مجاني من يوتيوب
    #   - لا يحتاج مفتاح API
    #   - يرجع معلومات أساسية (العنوان، صاحب القناة، الصورة المصغرة)
    #
    # الرابط:
    #   https://www.youtube.com/oembed?url=VIDEO_URL&format=json
    
    if not REQUESTS_AVAILABLE:
        return None
    
    try:
        oembed_url = f"https://www.youtube.com/oembed?url={url}&format=json"
        response = requests.get(oembed_url, timeout=10)
        # ↑ إرسال طلب HTTP مع timeout 10 ثواني
        
        if response.status_code == 200:
            data = response.json()
            return data.get('title')
            # ↑ استخراج العنوان من الاستجابة
        
        return None
        
    except Exception as e:
        log(f"خطأ في جلب عنوان الفيديو: {e}", "WARNING")
        return None


def fetch_youtube_playlist_info(url: str) -> Dict[str, Any]:
    """
    جلب معلومات قائمة تشغيل يوتيوب
    """
    # ─────────────────────────────────────────────────────────────────────────
    # يستخدم yt-dlp لجلب:
    #   - عنوان القائمة
    #   - عدد الفيديوهات
    #   - عدد المشاهدات (إن توفر)
    
    result = {
        'title': '',
        'videoCount': 0,
        'totalViews': 0,
    }
    
    if not YT_DLP_AVAILABLE:
        return result
    
    try:
        ydl_opts = {
            'quiet': True,           # عدم طباعة رسائل
            'no_warnings': True,     # عدم طباعة تحذيرات
            'extract_flat': True,    # استخراج معلومات بدون تحميل
            'skip_download': True,   # عدم تحميل الفيديوهات
        }
        
        with YoutubeDL(ydl_opts) as ydl:
            info = ydl.extract_info(url, download=False)
            # ↑ استخراج المعلومات فقط بدون تحميل
            
            if info:
                result['title'] = info.get('title', '')
                entries = info.get('entries', [])
                result['videoCount'] = len(entries) if entries else 0
                result['totalViews'] = info.get('view_count', 0)
        
        return result
        
    except Exception as e:
        log(f"خطأ في جلب معلومات القائمة: {e}", "WARNING")
        return result


def fetch_webpage_title(url: str) -> Optional[str]:
    """
    جلب عنوان صفحة ويب عادية
    """
    # ─────────────────────────────────────────────────────────────────────────
    # يستخدم:
    #   - requests لجلب الصفحة
    #   - BeautifulSoup لتحليل HTML
    #
    # يبحث عن:
    #   1. عنصر <title>
    #   2. meta property="og:title"
    
    if not REQUESTS_AVAILABLE or not BS4_AVAILABLE:
        return None
    
    try:
        headers = {
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
        }
        # ↑ تقمص متصفح لتجنب الحظر
        
        response = requests.get(url, headers=headers, timeout=10)
        
        if response.status_code == 200:
            soup = BeautifulSoup(response.text, 'html.parser')
            # ↑ تحليل HTML
            
            title_tag = soup.find('title')
            if title_tag:
                return title_tag.get_text().strip()
            
            og_title = soup.find('meta', property='og:title')
            if og_title:
                return og_title.get('content', '').strip()
        
        return None
        
    except Exception as e:
        log(f"خطأ في جلب عنوان الصفحة: {e}", "WARNING")
        return None

def load_entries_from_json(file_path: Path) -> List[Dict]:
    """
    قراءة الإدخالات من ملف JSON
    """
    # ─────────────────────────────────────────────────────────────────────────
    # يدعم صيغتين:
    #   1. مصفوفة مباشرة: [{...}, {...}]
    #   2. كائن مع entries: {"entries": [{...}, {...}]}
    
    if not file_path.exists():
        return []
    
    try:
        with file_path.open('r', encoding='utf-8') as f:
            data = json.load(f)
        
        if isinstance(data, list):
            return data
        elif isinstance(data, dict) and 'entries' in data:
            return data['entries']
        else:
            return []
            
    except json.JSONDecodeError as e:
        log(f"خطأ في قراءة ملف JSON: {e}", "ERROR")
        return []


def save_entries_to_json(entries: List[Dict], file_path: Path) -> None:
    """
    حفظ الإدخالات في ملف JSON
    """
    # ─────────────────────────────────────────────────────────────────────────
    # يحفظ بالصيغة:
    # {
    #   "generatedAt": "2024-01-15T14:30:45.123456",
    #   "total": 100,
    #   "entries": [...]
    # }
    
    file_path.parent.mkdir(parents=True, exist_ok=True)
    # ↑ إنشاء المجلدات إذا لم تكن موجودة
    
    payload = {
        'generatedAt': dt.datetime.now().isoformat(),
        'total': len(entries),
        'entries': entries,
    }
    
    with file_path.open('w', encoding='utf-8') as f:
        json.dump(payload, f, ensure_ascii=False, indent=2)
    # ↑ ensure_ascii=False: للحفاظ على الأحرف العربية
    # ↑ indent=2: تنسيق مقروء
    
    log(f"تم حفظ {len(entries)} إدخال في {file_path}", "SUCCESS")

def build_ydl_opts(output_dir: Path, fmt: str, write_metadata: bool = True) -> Dict:
    """
    بناء خيارات yt-dlp
    """
    # ─────────────────────────────────────────────────────────────────────────
    # الإعدادات الرئيسية:
    
    outtmpl = str(output_dir / "%(uploader)s/%(upload_date)s - %(title)s [%(id)s].%(ext)s")
    # ↑ نمط اسم الملف:
    #   archive/ChannelName/20240115 - Video Title [abc123xyz].mp4
    
    ydl_opts: Dict = {
        'outtmpl': outtmpl,           # نمط اسم الملف
        'format': fmt,                 # صيغة التحميل
        'noplaylist': False,           # السماح بتحميل قوائم التشغيل
        'ignoreerrors': True,          # تجاهل الأخطاء والاستمرار
        'retries': 5,                  # عدد محاولات إعادة المحاولة
        'fragment_retries': 10,        # محاولات الأجزاء
        'concurrent_fragment_downloads': 5,  # تحميل متوازي للأجزاء
        'merge_output_format': 'mp4',  # دمج في MP4
        'continuedl': True,            # استكمال التحميل
        'noprogress': False,           # إظهار شريط التقدم
    }
    
    if write_metadata:
        ydl_opts.update({
            'writeinfojson': True,     # حفظ معلومات JSON
            'writesubtitles': True,    # حفظ الترجمات
            'writeautomaticsub': True, # حفظ الترجمات التلقائية
            'writethumbnail': True,    # حفظ الصورة المصغرة
        })
    
    return ydl_opts


def download_all(entries, output_dir, fmt, max_concurrent, log_file, write_metadata):
    """
    تحميل جميع الفيديوهات بالتوازي
    """
    # ─────────────────────────────────────────────────────────────────────────
    # يستخدم ThreadPoolExecutor للتحميل المتوازي
    # يفلتر الروابط القابلة للتحميل (يوتيوب فقط)
    # ينشئ مجلدات فرعية حسب التصنيف (shorts, videos, إلخ)
    
    with concurrent.futures.ThreadPoolExecutor(max_workers=max_concurrent) as executor:
        futures = {}
        
        for i, entry in enumerate(downloadable, 1):
            folder = entry.get('folder', 'videos')
            entry_output = output_dir / folder
            # ↑ إنشاء مجلد فرعي حسب التصنيف
            
            future = executor.submit(
                download_one, i, url, entry_output, fmt, log_file, write_metadata
            )
            futures[future] = url
        
        for future in concurrent.futures.as_completed(futures):
            if future.result():
                success += 1
            else:
                failed += 1
